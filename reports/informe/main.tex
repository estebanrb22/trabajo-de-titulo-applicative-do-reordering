% Parámetros principales de los nodos: #1 largo, #2 alto y #3 fuente interna.
\newcommand{\GraphNodeFontSize}{}
\newcommand{\setgraphnodesize}[3]{%
    \setlength{\GraphNodeWidth}{#1}%
    \setlength{\GraphNodeHeight}{#2}%
    \renewcommand{\GraphNodeFontSize}{#3}%
}

\setgraphnodesize
    {2.3cm} % Largo del óvalo
    {1.3cm} % Alto del óvalo
    {9pt}   % Tamaño de la fuente interna

\setlength{\GraphNodeInnerXSep}{1pt}
\setlength{\GraphNodeInnerYSep}{1pt}
\setlength{\GraphHorizontalSeparation}{0.75cm}
\setlength{\GraphVerticalSeparation}{0.9cm}
\setlength{\GraphLabelDistance}{3pt}
\setlength{\GraphNodeBorderWidth}{0.5pt}
\setlength{\GraphEdgeWidth}{0.8pt}

\newcommand{\GraphNodeFont}{%
    \ttfamily\fontsize{\GraphNodeFontSize}{\GraphNodeFontSize}\selectfont
}
\newcommand{\GraphLabelFont}{\small}
\newcommand{\GraphLongEdgeBend}{22}

\tikzset{
    precedence node/.style={
        draw=graphNodeBorder,
        fill=graphNodeBg,
        text=graphNodeFg,
        line width=\GraphNodeBorderWidth,
        ellipse,
        minimum width=\GraphNodeWidth,
        minimum height=\GraphNodeHeight,
        inner xsep=\GraphNodeInnerXSep,
        inner ysep=\GraphNodeInnerYSep,
        label distance=\GraphLabelDistance,
        align=center,
        font=\GraphNodeFont
    },
    precedence label/.style={
        text=graphLabelFg,
        font=\GraphLabelFont
    },
    precedence edge/.style={
        ->,
        >=stealth,
        draw=graphEdge,
        line width=\GraphEdgeWidth
    }
}

\newenvironment{PrecedenceGraph}[1][]{%
    \begin{tikzpicture}[
        node distance=\GraphVerticalSeparation and \GraphHorizontalSeparation,
        #1
    ]%
}{%
    \end{tikzpicture}%
}

\newcommand{\PrecedenceNode}[4][]{%
    \node[precedence node,label={[precedence label]below:{#4}},#1] (#2) {#3};%
}

\newcommand{\PrecedenceEdge}[3][]{%
    \draw[precedence edge,#1] (#2) to (#3);%
}
